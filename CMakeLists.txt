cmake_minimum_required(VERSION 3.10)
project(StudentRankingSystem CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE) set(CMAKE_BUILD_TYPE Release) endif()
find_package(Threads REQUIRED)
add_executable(ranking_server backend/src/main.cpp backend/src/StudentManager.cpp)
target_include_directories(ranking_server PRIVATE backend/include third_party)
target_link_libraries(ranking_server PRIVATE Threads::Threads)
if(WIN32) target_link_libraries(ranking_server PRIVATE ws2_32) endif()
